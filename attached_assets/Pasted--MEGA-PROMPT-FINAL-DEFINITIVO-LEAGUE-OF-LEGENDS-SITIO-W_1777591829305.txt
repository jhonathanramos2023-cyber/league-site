### ChampionCard (componente):
```jsx
const ChampionCard = ({ champion, index }) => {
  const { ref, inView } = useInView({ threshold: 0.1, triggerOnce: true });
  
  // Imagen: SIEMPRE loading screen art (composición vertical perfecta)
  const imgURL = `https://ddragon.leagueoflegends.com/cdn/img/champion/loading/${champion.id}_0.jpg`;
  
  return (
    <motion.div
      ref={ref}
      initial={{ opacity: 0, y: 60 }}
      animate={inView ? { opacity: 1, y: 0 } : {}}
      transition={{ duration: 0.5, delay: (index % 6) * 0.07 }}
      onMouseMove={handle3DTilt}
      onMouseLeave={resetTilt}
      style={{ transformStyle: 'preserve-3d', perspective: '1000px' }}
      onClick={() => navigate(`/campeon/${champion.id}`)}
    >
      <img
        src={imgURL}
        alt={getChampionName(champion.id)}
        loading="lazy"
        style={{ objectFit: 'cover', objectPosition: 'center 15%' }}
      />
      {/* Badge de rol */}
      {/* Nombre: getChampionName(champion.id) — NUNCA champion.name */}
      {/* Título en español */}
      {/* Overlay con botón al hover */}
    </motion.div>
  );
};

// EFECTO 3D TILT al hover:
const handle3DTilt = (e) => {
  const rect = e.currentTarget.getBoundingClientRect();
  const x = (e.clientX - rect.left) / rect.width - 0.5;
  const y = (e.clientY - rect.top) / rect.height - 0.5;
  e.currentTarget.style.transform = `
    perspective(1000px) rotateY(${x * 12}deg) rotateX(${-y * 12}deg) translateZ(8px)
  `;
};
```

### Cursor Custom (reemplaza el cursor del sistema):
```jsx
const CustomCursor = () => {
  const cursorDot = useRef(null);
  const cursorRing = useRef(null);
  
  useEffect(() => {
    const moveCursor = (e) => {
      cursorDot.current.style.left = e.clientX + 'px';
      cursorDot.current.style.top = e.clientY + 'px';
      gsap.to(cursorRing.current, { left: e.clientX, top: e.clientY, duration: 0.15 });
    };
    window.addEventListener('mousemove', moveCursor);
    return () => window.removeEventListener('mousemove', moveCursor);
  }, []);
  
  return (
    <>
      {/* Punto dorado pequeño — instantáneo */}
      <div ref={cursorDot} style={{
        position: 'fixed', width: 6, height: 6, borderRadius: '50%',
        background: '#C8AA6E', pointerEvents: 'none', zIndex: 9999,
        transform: 'translate(-50%, -50%)'
      }}/>
      {/* Anillo que sigue con delay */}
      <div ref={cursorRing} style={{
        position: 'fixed', width: 28, height: 28, borderRadius: '50%',
        border: '1.5px solid #C8AA6E', pointerEvents: 'none', zIndex: 9998,
        transform: 'translate(-50%, -50%)', opacity: 0.6
      }}/>
    </>
  );
};
```
